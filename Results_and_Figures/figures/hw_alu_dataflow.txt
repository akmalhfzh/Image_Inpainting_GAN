=============================================================================
CONV_UNIFIED - DEEP ALU-LEVEL DATAPATH DIAGRAM
=============================================================================
Catatan: Ini adalah rincian level gerbang/ALU paling bawah dari conv_unified. 
Setiap operasi ditandai dengan [Tipe Hardware] yang di-infer oleh sintesis FPGA.
Index 'm' merepresentasikan 16 jalur paralel (m = 0 sampai 15).

-----------------------------------------------------------------------------
1. ADDRESS GENERATOR (addr_gen.v) -> Dijalankan 16x secara paralel
-----------------------------------------------------------------------------
// Y-Coordinate Math
LAYER_CONFIG --(h_out)--> MULTIPLIER_Y [Tipe: Multiplier]
LAYER_CONFIG --(stride)--> MULTIPLIER_Y [Tipe: Multiplier]
MULTIPLIER_Y --(base_y)--> SUBTRACTOR_Y [Tipe: Subtractor]
LAYER_CONFIG --(pad)--> SUBTRACTOR_Y [Tipe: Subtractor]
SUBTRACTOR_Y --(base_y_pad)--> ADDER_IN_Y [Tipe: Adder]
FSM_CONTROL --(dy)--> ADDER_IN_Y [Tipe: Adder]

// X-Coordinate Math
LAYER_CONFIG --(w_out)--> MULTIPLIER_X [Tipe: Multiplier]
LAYER_CONFIG --(stride)--> MULTIPLIER_X [Tipe: Multiplier]
MULTIPLIER_X --(base_x)--> SUBTRACTOR_X [Tipe: Subtractor]
LAYER_CONFIG --(pad)--> SUBTRACTOR_X [Tipe: Subtractor]
SUBTRACTOR_X --(base_x_pad)--> ADDER_IN_X [Tipe: Adder]
FSM_CONTROL --(dx)--> ADDER_IN_X [Tipe: Adder]

// Boundary/Padding Check
ADDER_IN_Y --(in_y)--> COMPARATOR_Y1 [Tipe: Comparator >=]
CONSTANT --(0)--> COMPARATOR_Y1 [Tipe: Comparator >=]
ADDER_IN_Y --(in_y)--> COMPARATOR_Y2 [Tipe: Comparator <]
LAYER_CONFIG --(h_in)--> COMPARATOR_Y2 [Tipe: Comparator <]

ADDER_IN_X --(in_x)--> COMPARATOR_X1 [Tipe: Comparator >=]
CONSTANT --(0)--> COMPARATOR_X1 [Tipe: Comparator >=]
ADDER_IN_X --(in_x)--> COMPARATOR_X2 [Tipe: Comparator <]
LAYER_CONFIG --(w_in)--> COMPARATOR_X2 [Tipe: Comparator <]

COMPARATOR_Y1 --(y1_res)--> AND_GATE_1 [Tipe: Logic AND]
COMPARATOR_Y2 --(y2_res)--> AND_GATE_1 [Tipe: Logic AND]
COMPARATOR_X1 --(x1_res)--> AND_GATE_2 [Tipe: Logic AND]
COMPARATOR_X2 --(x2_res)--> AND_GATE_2 [Tipe: Logic AND]

AND_GATE_1 --(is_y_valid)--> AND_GATE_FINAL [Tipe: Logic AND]
AND_GATE_2 --(is_x_valid)--> AND_GATE_FINAL [Tipe: Logic AND]
AND_GATE_FINAL --(valid_vec[m])--> PIPE_REG_ADDR [Tipe: D-FlipFlop]

// Final 1D Memory Offset
ADDER_IN_Y --(in_y)--> MULTIPLIER_OFFSET [Tipe: Multiplier]
LAYER_CONFIG --(w_in)--> MULTIPLIER_OFFSET [Tipe: Multiplier]
MULTIPLIER_OFFSET --(in_y * w_in)--> ADDER_ADDR [Tipe: Adder]
ADDER_IN_X --(in_x)--> ADDER_ADDR [Tipe: Adder]
ADDER_ADDR --(ifmap_addrs[m])--> PIPE_REG_ADDR [Tipe: D-FlipFlop]


-----------------------------------------------------------------------------
2. MAC ARRAY (mac_array.v) -> Komputasi Utama
-----------------------------------------------------------------------------
// Stage 0: 16x Parallel Multipliers (16-bit x 16-bit)
IFMAP_TILE_REG --(ifmap_tile[m])--> ZERO_PAD_MUX [Tipe: 2-to-1 Multiplexer]
PIPE_REG_ADDR --(valid_vec[m])--> ZERO_PAD_MUX (Sebagai sinyal Select)
CONSTANT --(16'd0)--> ZERO_PAD_MUX [Tipe: 2-to-1 Multiplexer]
ZERO_PAD_MUX --(ival)--> MAC_MULTIPLIER [Tipe: Signed Multiplier]

WEIGHT_CACHE_BRAM --(weight_tile[m])--> MAC_MULTIPLIER [Tipe: Signed Multiplier]
MAC_MULTIPLIER --(prod[m]: 32-bit)--> SIGN_EXTEND_LOGIC [Tipe: Wire routing]
SIGN_EXTEND_LOGIC --(prod_sx[m]: 48-bit)--> PIPE_REG_MAC1 [Tipe: D-FlipFlop]

// Stage 1: Binary Adder Tree (adder_tree.v)
PIPE_REG_MAC1 --(prod_sx[0], prod_sx[1])--> ADDER_L1_0 [Tipe: 48-bit Adder]
PIPE_REG_MAC1 --(prod_sx[2], prod_sx[3])--> ADDER_L1_1 [Tipe: 48-bit Adder]
... (Total 8 Adders di Level 1)

ADDER_L1_0 --(sum_L1_0)--> ADDER_L2_0 [Tipe: 48-bit Adder]
ADDER_L1_1 --(sum_L1_1)--> ADDER_L2_0 [Tipe: 48-bit Adder]
... (Total 4 Adders di Level 2)

ADDER_L2_0 --(sum_L2_0)--> ADDER_L3_0 [Tipe: 48-bit Adder]
ADDER_L2_1 --(sum_L2_1)--> ADDER_L3_0 [Tipe: 48-bit Adder]
... (Total 2 Adders di Level 3)

ADDER_L3_0 --(sum_L3_0)--> ADDER_L4_FINAL [Tipe: 48-bit Adder]
ADDER_L3_1 --(sum_L3_1)--> ADDER_L4_FINAL [Tipe: 48-bit Adder]

ADDER_L4_FINAL --(row_sum_comb)--> PIPE_REG_MAC2 [Tipe: D-FlipFlop]
PIPE_REG_MAC2 --(row_sum: 48-bit)--> ACCUMULATOR_BLOCK


-----------------------------------------------------------------------------
3. ACCUMULATOR & REQUANTIZATION (Di dalam conv_unified.v)
-----------------------------------------------------------------------------
// Bias Addition & Accumulation
BIAS_MEM_BRAM --(bias_val)--> BIAS_SIGN_EXT [Tipe: Wire routing]
BIAS_SIGN_EXT --(bias_extended)--> BIAS_ACC_MUX [Tipe: 2-to-1 Multiplexer]
ACCUMULATOR_REG --(acc_reg)--> BIAS_ACC_MUX [Tipe: 2-to-1 Multiplexer]
FSM_CONTROL --(is_first_channel)--> BIAS_ACC_MUX (Sebagai sinyal Select)

BIAS_ACC_MUX --(bias_or_acc)--> ACC_ADDER [Tipe: 48-bit Adder]
PIPE_REG_MAC2 --(row_sum)--> ACC_ADDER [Tipe: 48-bit Adder]
ACC_ADDER --(acc_next)--> ACCUMULATOR_REG [Tipe: D-FlipFlop]

// Requantization (Rounding & Saturation)
ACCUMULATOR_REG --(acc_reg)--> ROUNDING_ADDER [Tipe: 48-bit Adder]
CONSTANT --(0.5 ULP)--> ROUNDING_ADDER [Tipe: 48-bit Adder]
ROUNDING_ADDER --(acc_rounded)--> RIGHT_SHIFTER [Tipe: Arithmetic Shift Right]
RIGHT_SHIFTER --(acc_shifted: 40-bit)--> SAT_MAX_CMP [Tipe: Comparator >]
CONSTANT --(+32767)--> SAT_MAX_CMP [Tipe: Comparator >]

RIGHT_SHIFTER --(acc_shifted: 40-bit)--> SAT_MIN_CMP [Tipe: Comparator <]
CONSTANT --(-32768)--> SAT_MIN_CMP [Tipe: Comparator <]

SAT_MAX_CMP --(sat_pos)--> SAT_MUX_1 [Tipe: 2-to-1 Multiplexer]
SAT_MIN_CMP --(sat_neg)--> SAT_MUX_1 (Sebagai sinyal Select)
RIGHT_SHIFTER --(acc_shifted)--> SAT_MUX_1 [Tipe: 2-to-1 Multiplexer]
CONSTANT --(-32768)--> SAT_MUX_1 [Tipe: 2-to-1 Multiplexer]

SAT_MUX_1 --(mux1_out)--> SAT_MUX_2 [Tipe: 2-to-1 Multiplexer]
SAT_MAX_CMP --(sat_pos)--> SAT_MUX_2 (Sebagai sinyal Select)
CONSTANT --(+32767)--> SAT_MUX_2 [Tipe: 2-to-1 Multiplexer]

SAT_MUX_2 --(acc_trunc: 16-bit)--> ACTIVATION_UNITS


-----------------------------------------------------------------------------
4. ACTIVATION FUNCTIONS (Parallel Hardware)
-----------------------------------------------------------------------------
// --- Leaky ReLU ---
SAT_MUX_2 --(acc_trunc)--> LRELU_CMP [Tipe: Comparator >=]
CONSTANT --(0)--> LRELU_CMP [Tipe: Comparator >=]

SAT_MUX_2 --(acc_trunc)--> LRELU_MULT [Tipe: Signed Multiplier]
CONSTANT --(51)--> LRELU_MULT [Tipe: Signed Multiplier]
LRELU_MULT --(mult_res)--> LRELU_SHIFT [Tipe: Arithmetic Shift Right]

LRELU_SHIFT --(neg_val)--> LRELU_MUX [Tipe: 2-to-1 Multiplexer]
SAT_MUX_2 --(acc_trunc)--> LRELU_MUX [Tipe: 2-to-1 Multiplexer]
LRELU_CMP --(is_pos)--> LRELU_MUX (Sebagai sinyal Select)
LRELU_MUX --(lrelu_out)--> FINAL_ACT_MUX

// --- ReLU ---
SAT_MUX_2 --(acc_trunc)--> RELU_CMP [Tipe: Comparator >=]
CONSTANT --(0)--> RELU_CMP [Tipe: Comparator >=]

SAT_MUX_2 --(acc_trunc)--> RELU_MUX [Tipe: 2-to-1 Multiplexer]
CONSTANT --(0)--> RELU_MUX [Tipe: 2-to-1 Multiplexer]
RELU_CMP --(is_pos)--> RELU_MUX (Sebagai sinyal Select)
RELU_MUX --(relu_out)--> FINAL_ACT_MUX

// --- Tanh (Simplified) ---
SAT_MUX_2 --(acc_trunc)--> TANH_ABS_SUB [Tipe: Subtractor] (Menghitung Absolute)
TANH_ABS_SUB --(abs_x)--> TANH_LUT_BRAM [Tipe: Memory Read]
TANH_ABS_SUB --(abs_x)--> TANH_DIFF_SUB [Tipe: Subtractor]
TANH_DIFF_SUB --(diff)--> TANH_INTERP_MULT [Tipe: Multiplier]
TANH_INTERP_MULT --(interp_val)--> TANH_ADDER [Tipe: Adder]
TANH_LUT_BRAM --(base_val)--> TANH_ADDER [Tipe: Adder]
TANH_ADDER --(tanh_mag)--> TANH_SIGN_SUB [Tipe: Subtractor] (Mengembalikan Tanda Minus)
TANH_SIGN_SUB --(tanh_out)--> FINAL_ACT_MUX


-----------------------------------------------------------------------------
5. OUTPUT MUX
-----------------------------------------------------------------------------
LRELU_MUX --(lrelu_out)--> FINAL_ACT_MUX [Tipe: 4-to-1 Multiplexer]
RELU_MUX --(relu_out)--> FINAL_ACT_MUX [Tipe: 4-to-1 Multiplexer]
TANH_SIGN_SUB --(tanh_out)--> FINAL_ACT_MUX [Tipe: 4-to-1 Multiplexer]
SAT_MUX_2 --(acc_trunc)--> FINAL_ACT_MUX [Tipe: 4-to-1 Multiplexer] (Bypass Path)

LAYER_CONFIG --(cur_act)--> FINAL_ACT_MUX (Sebagai sinyal Select 2-bit)
FINAL_ACT_MUX --(final_pixel)--> AXI_WRITE_LOGIC
